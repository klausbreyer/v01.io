{{- $destination := .Destination | safeURL -}} {{- $imageSrc := $destination -}} {{- $page := .Page -}} {{- /* Check if destination is a relative path (not starting with / or http) */ -}} {{- if and (not (hasPrefix $destination "/")) (not (hasPrefix $destination "http")) -}} {{- /* Try to find image in page resources */ -}} {{- $imageName := path.Base $destination -}} {{- $resource := $page.Resources.GetMatch $imageName -}} {{- if not $resource -}} {{- $resource = resources.Get (printf "%s%s" $page.File.Dir $imageName) -}} {{- end -}} {{- if $resource -}} {{- $imageSrc = $resource.RelPermalink -}} {{- else -}} {{- /* Preserve the original path when no local resource exists. */ -}} {{- $imageSrc = $destination -}} {{- end -}} {{- end -}}
